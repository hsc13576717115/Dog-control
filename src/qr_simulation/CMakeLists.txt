cmake_minimum_required(VERSION 3.16)
project(qr_simulation)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(gazebo REQUIRED)
find_package(gazebo_ros REQUIRED)
find_package(qr_interfaces REQUIRED)
add_library(qr_contact_metrics SHARED src/ContactMetricsPlugin.cpp)
target_include_directories(qr_contact_metrics PRIVATE ${GAZEBO_INCLUDE_DIRS})
target_link_libraries(qr_contact_metrics ${GAZEBO_LIBRARIES})
ament_target_dependencies(qr_contact_metrics gazebo_ros qr_interfaces)
install(TARGETS qr_contact_metrics LIBRARY DESTINATION lib)
ament_package()
