cmake_minimum_required(VERSION 3.16)
project(custom_dog_rl LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(CUSTOM_DOG_RL_BUILD_ROS "Build the ROS 2 controller plugin" ON)
set(ONNXRUNTIME_ROOT "" CACHE PATH "Extracted ONNX Runtime C/C++ release directory")
set(CUSTOM_DOG_RL_TEST_MODEL "${CMAKE_CURRENT_SOURCE_DIR}/models/terrain_16099/policy_combined.onnx" CACHE FILEPATH "Verified model used by inference tests")
include(CTest)
find_package(Threads REQUIRED)
find_path(ONNXRUNTIME_INCLUDE_DIR onnxruntime_cxx_api.h HINTS "${ONNXRUNTIME_ROOT}/include" REQUIRED)
find_library(ONNXRUNTIME_LIBRARY onnxruntime HINTS "${ONNXRUNTIME_ROOT}/lib" REQUIRED)
add_library(custom_dog_rl_core SHARED src/PolicyCore.cpp src/OnnxPolicy.cpp)
target_include_directories(custom_dog_rl_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_include_directories(custom_dog_rl_core SYSTEM PRIVATE "${ONNXRUNTIME_INCLUDE_DIR}")
target_link_libraries(custom_dog_rl_core PRIVATE "${ONNXRUNTIME_LIBRARY}" Threads::Threads)
# Preserve the runtime lookup after installation; NX uses the SDK installed by
# scripts/install_onnxruntime.sh. No x86 runtime binary is shipped in the package.
get_filename_component(ONNXRUNTIME_LIBRARY_DIR "${ONNXRUNTIME_LIBRARY}" DIRECTORY)
set_target_properties(custom_dog_rl_core PROPERTIES INSTALL_RPATH "${ONNXRUNTIME_LIBRARY_DIR}")
target_compile_options(custom_dog_rl_core PRIVATE -Wall -Wextra -Wpedantic)

add_executable(policy_probe tools/policy_probe.cpp)
target_link_libraries(policy_probe custom_dog_rl_core)

if(BUILD_TESTING)
  add_executable(test_policy_core test/test_policy_core.cpp)
  target_link_libraries(test_policy_core custom_dog_rl_core)
  add_test(NAME policy_core COMMAND test_policy_core)
  if(EXISTS "${CUSTOM_DOG_RL_TEST_MODEL}")
    add_executable(test_onnx_policy test/test_onnx_policy.cpp)
    target_link_libraries(test_onnx_policy custom_dog_rl_core)
    add_test(NAME onnx_policy COMMAND test_onnx_policy "${CUSTOM_DOG_RL_TEST_MODEL}")
  else()
    message(STATUS "Model-dependent tests skipped: set CUSTOM_DOG_RL_TEST_MODEL to the verified combined ONNX file")
  endif()
endif()

if(CUSTOM_DOG_RL_BUILD_ROS)
  find_package(ament_cmake REQUIRED)
  find_package(controller_interface REQUIRED)
  find_package(hardware_interface REQUIRED)
  find_package(geometry_msgs REQUIRED)
  find_package(pluginlib REQUIRED)
  find_package(rclcpp REQUIRED)
  find_package(rclcpp_lifecycle REQUIRED)
  find_package(realtime_tools REQUIRED)
  find_package(sensor_msgs REQUIRED)
  find_package(std_msgs REQUIRED)
  find_package(OpenSSL REQUIRED)
  add_library(custom_dog_rl_controller SHARED src/RlController.cpp)
  target_link_libraries(custom_dog_rl_controller custom_dog_rl_core OpenSSL::Crypto Threads::Threads)
  target_compile_options(custom_dog_rl_controller PRIVATE -Wall -Wextra -Wpedantic)
  ament_target_dependencies(custom_dog_rl_controller controller_interface hardware_interface
    geometry_msgs pluginlib rclcpp rclcpp_lifecycle realtime_tools sensor_msgs std_msgs)
  pluginlib_export_plugin_description_file(controller_interface custom_dog_rl_plugin.xml)
  install(TARGETS policy_probe RUNTIME DESTINATION lib/${PROJECT_NAME})
  install(TARGETS custom_dog_rl_core custom_dog_rl_controller EXPORT export_custom_dog_rl
    ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
  install(DIRECTORY include/ DESTINATION include)
  install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
  install(PROGRAMS scripts/mock_imu.py DESTINATION lib/${PROJECT_NAME})
  install(FILES custom_dog_rl_plugin.xml DESTINATION share/${PROJECT_NAME})
  if(BUILD_TESTING AND EXISTS "${CUSTOM_DOG_RL_TEST_MODEL}" AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/test/test_rl_controller.cpp")
    find_package(ament_cmake_gtest REQUIRED)
    ament_add_gtest(test_rl_controller test/test_rl_controller.cpp TIMEOUT 90)
    set_tests_properties(test_rl_controller PROPERTIES ENVIRONMENT "ROS_DOMAIN_ID=83")
    target_link_libraries(test_rl_controller custom_dog_rl_controller)
    target_compile_definitions(test_rl_controller PRIVATE CUSTOM_DOG_RL_TEST_MODEL="${CUSTOM_DOG_RL_TEST_MODEL}")
    ament_target_dependencies(test_rl_controller controller_interface hardware_interface
      geometry_msgs rclcpp rclcpp_lifecycle realtime_tools sensor_msgs std_msgs)
  endif()
  ament_export_targets(export_custom_dog_rl HAS_LIBRARY_TARGET)
  ament_export_dependencies(controller_interface hardware_interface geometry_msgs pluginlib rclcpp
    rclcpp_lifecycle realtime_tools sensor_msgs std_msgs)
  ament_package()
else()
  install(TARGETS custom_dog_rl_core LIBRARY DESTINATION lib)
  install(TARGETS policy_probe RUNTIME DESTINATION bin)
  install(DIRECTORY include/ DESTINATION include)
endif()
